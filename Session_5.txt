CREATE TABLE influencers (
    influencer_id INT PRIMARY KEY,
    name VARCHAR(100)
);

CREATE TABLE posts (
    post_id INT PRIMARY KEY,
    influencer_id INT,
    caption VARCHAR(255),
    FOREIGN KEY (influencer_id) REFERENCES influencers(influencer_id)
);

5.2
SELECT posts.caption, influencers.name
FROM posts
INNER JOIN influencers
ON posts.influencer_id = influencers.influencer_id;

5.3
SELECT influencers.name, COALESCE(posts.caption, 'No Posts') AS caption
FROM influencers
LEFT JOIN posts
ON influencers.influencer_id = posts.influencer_id;

5.4
SELECT posts.caption, influencers.name
FROM influencers
RIGHT JOIN posts
ON influencers.influencer_id = posts.influencer_id;
